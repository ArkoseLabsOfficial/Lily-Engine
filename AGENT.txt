    private function importFailedCallback(cl:Array<String>, ?asName:String):Bool {
        var dotted:String = cl.join(".");
        var base:String = dotted.split(".").pop();

        if (Interp.scriptClasses.exists(base))
            return true;

        var relPath:String = dotted.replace(".", "/");
        var candidates:Array<String> = [];

        for (ext in Script.scriptExtensions) {
            candidates.push('scripts/$relPath.$ext');
            candidates.push('script/$relPath.$ext');
            candidates.push('$relPath.$ext');
            candidates.push('source/$relPath.$ext');
        }

        for (candidate in candidates) {
            if (__importedPaths.contains(candidate))
                continue;

            var fullPath:String = LilyAssets.getPath(candidate);
            if (FileSystem.exists(fullPath)) {
                var script:Script = Script.create(candidate);
                if (!(script is DummyScript)) {
                    __importedPaths.push(candidate);
                    script.ownerPack = ownerPack;
                    if (ownerPack != null && !ownerPack.contains(script.path))
                        ownerPack.add(script);
                    script.load();
                    return Interp.scriptClasses.exists(base);
                }
            }
        }

        return false;
    }

    private function _errorHandler(error:Error) {

        var fn = '$fileName:${error.line}: ';

        while (err.startsWith(oldfn) || err.startsWith(fn)) {
            if (err.startsWith(oldfn))
                err = err.substr(oldfn.length);
            if (err.startsWith(fn))
                err = err.substr(fn.length);

            sscript.interp.unregisterScriptScope();
            sscript.interp.allowStaticVariables = false;
            sscript.interp.allowPublicVariables = false;

            return null;

            && sscript.interp.publicVariables.exists(funcName))
            func = sscript.interp.publicVariables.get(funcName);
        else if (sscript.interp.allowStaticVariables && sscript.interp.staticVarNames.exists(funcName)
            && sscript.interp.staticVariables.exists(funcName))
            func = sscript.interp.staticVariables.get(funcName);
        else
            func = sscript.get(funcName);

        if (func != null && Reflect.isFunction(func))
            return Reflect.callMethod(null, func, parameters);

        return null;
    }

        if (p.startsWith("source/"))
            p = p.substr(7);

        p = p.replace("/", ".");

        if (p == "")
            p = "Script";

        return p;

@:access(CancellableEvent)
class ScriptPack extends Script {
    public var scripts:Array<Script> = [];
    public var additionalDefaultVariables:Map<String, Dynamic> = [];
    public var publicVariables:Map<String, Dynamic> = [];
    public var parent:Dynamic = null;

    public override function load() {
        for (e in scripts)
            e.load();
    }

    public function contains(path:String) {
        for (e in scripts)
            if (e.path == path)
                return true;
        return false;
    }

    public function new(name:String) {
        additionalDefaultVariables["importScript"] = importScript;
        super(name);
        packScope = new ScriptPackScope(this);
        set("__script__", packScope);
    }

    public override function getScriptHandle():Dynamic {
        return packScope;
    }

    public function getByPath(name:String) {
        for (s in scripts)
            if (s.path == name)
                return s;
        return null;
    }

    public function getByName(name:String) {
        for (s in scripts)
            if (s.fileName == name)
                return s;
        return null;
    }

    public function importScript(path:String):Dynamic {
        var resolved:String = HScript.importScriptPathHandler != null ? HScript.importScriptPathHandler(path) : 'scripts/$path';
        if (resolved.indexOf(".") == -1)
            resolved += ".hx";

        var script = Script.create(resolved);
        if (script is DummyScript)
            throw 'Script at ${path} does not exist.';

        add(script);
        script.load();

        if (script is HScript) {
            var hs:HScript = cast script;
            if (hs.sscript != null && hs.sscript.interp.scriptScope != null)
                return hs.sscript.interp.scriptScope;
            return Interp.scriptClasses.get(hs.baseName);
        }
        return script;
    }

    public override function call(func:String, ?parameters:Array<Dynamic>):Dynamic {
        for (e in scripts)
            if (e.active)
                e.call(func, parameters);
        return null;
    }

    override public function event<T:CancellableEvent>(func:String, event:T):T {
        for (e in scripts) {
            if (!e.active)
                continue;
            e.call(func, [event]);
            if (event.cancelled && !event.__continueCalls)
                break;
        }
        return event;
    }

    public override function get(val:String):Dynamic {
        if (publicVariables.exists(val))
            return publicVariables.get(val);
        if (staticVariables.exists(val))
            return staticVariables.get(val);
        for (e in scripts) {
            var v = e.get(val);
            if (v != null)
                return v;
        }
        return null;
    }

    public override function reload() {
        for (e in scripts)
            e.reload();
    }

    public override function set(val:String, value:Dynamic) {
        publicVariables.set(val, value);
        for (e in scripts)
            e.set(val, value);
    }

    public override function setParent(parent:Dynamic) {
        this.parent = parent;
        for (e in scripts)
            e.setParent(parent);
    }

    public override function destroy() {
        super.destroy();
        for (e in scripts)
            e.destroy();
    }

    public override function onCreate(path:String) {}

    public function add(script:Script) {
        scripts.push(script);
        __configureNewScript(script);
    }

    public function remove(script:Script) {
        scripts.remove(script);
    }

    public function insert(pos:Int, script:Script) {
        scripts.insert(pos, script);
        __configureNewScript(script);
    }

    private function __configureNewScript(script:Script) {
        script.ownerPack = this;
        if (parent != null)
            script.setParent(parent);
        for (k => e in additionalDefaultVariables)
            script.set(k, e);
    }

    override public function toString():String {
        return FlxStringUtil.getDebugString([
            LabelValuePair.weak("parent", FlxStringUtil.getClassName(parent, true)),
            LabelValuePair.weak("total", scripts.length),
        ]);
    }

    public var staticVariables:Map<String, Dynamic> = [];
    public var packScope:ScriptPackScope;
}

    public function hget(f:String):Dynamic {
        if (f == "scripts") return pack.scripts;
        if (f == "publics") return pack.publicVariables;
        if (f == "statics") return pack.staticVariables;
        if (pack.publicVariables.exists(f)) return pack.publicVariables.get(f);
        if (pack.staticVariables.exists(f)) return pack.staticVariables.get(f);

        for (s in pack.scripts) {
            if (s is HScript) {
                var hs:HScript = cast s;
                if (hs.baseName == f) {
                    if (hs.sscript != null && hs.sscript.interp.scriptScope != null)
                        return hs.sscript.interp.scriptScope;
                    return Interp.scriptClasses.get(f);
                }
            }
        }
        return null;
    }

    public function hset(f:String, v:Dynamic):Dynamic {
        if (pack.publicVariables.exists(f)) {
            pack.publicVariables.set(f, v);
            return v;
        }
        if (pack.staticVariables.exists(f)) {
            pack.staticVariables.set(f, v);
            return v;
        }
        invalidAccess(f);
        return v;
    }

    function invalidAccess(f:String):Void {
        #if hscriptPos
        throw new Error(EInvalidAccess(f), 0, 0, pack.fileName, 0);
        #else
        throw EInvalidAccess(f);
        #end
    }

    public function __callGetter(f:String):Dynamic return hget(f);
    public function __callSetter(f:String, val:Dynamic):Dynamic return hset(f, val);
}
